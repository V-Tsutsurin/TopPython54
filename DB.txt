SELECT [ALL | DISTINCT] (* | столбец1 [,столбец-n])
FROM таблица1 [, таблица-n]
WHERE условие
			AND OR 
			выражение [NOT] BETWEEN начальное_значение AND конечное_значение
			выражение [NOT] LIKE шаблон_строки
				% - любое количество символов
				_ - один любой символ
			выражение [NOT] GLOB регулярное_выражение
				* - любое количество знаков
				? - соответствует одному символу
				. - соответствует любому одиночному символу
				[символы] - соответствует любому символу из списка [abc]
				[начальный-символ-конечный_символ] - соответствует любому одиночному символу из заданного диапозона [0-9]
				^ - этот символ используется в начале списка и соответствует любому символу, который НЕ вхлдит в список [^0-9]
			IS [NOT] NULL - позволяет выбирать все строки, столбцы которых имеют значение NULL
			выражение [NOT] IN (набор значений | выражение)
			EXISTS (существует)

Однострочные подзапросы: =, >, < , >=, <= (возвращает из подзапроса только одну строку)

многострочные подзапросы: IN (NOT IN) (возвращают более одной строки, список значений)

Операторы BETWEEN, LIKE, IS NULL нельзя применять к подзапросам

В подзапросах нельзя использовать ORDER BY, GROUP BY

FROM table_name_1 INNER JOIN table_name_2
ON связь_между_таблицами(первичным_и_внешним_ключами)

UNION - используется для обьединения двух и более SELECT с исключением повторяющихся строк

INNER JOIN - внутренне обьединение, включаются только те записи, которые имеют одинаковые значение в обееих обединенных таблицах

OUTER JOIN - внешнее обьединение

LEFT JOIN - используется для создания левого внешнего соединения. С помощью него выбираются все записи первой(левой) таблицы,
даже если они не соответствуют записям во второй(правой) таблице.

RIGHT JOIN - используется для создания правого внешнего соединения. С помощью него выбираются все записи второй(правой) таблицы,
даже если они не соответствуют записям в первой(левой) таблице.
